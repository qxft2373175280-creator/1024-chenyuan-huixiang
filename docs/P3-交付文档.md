> 本文档只包含 P3 的文件所有权范围，所有代码按 §4 契约编写，**不修改 `contracts.ts`**。
> 落地时按下面路径拆分文件即可；P1/P2 就绪后先跑 `npm run data:validate` 和 `npm run check`。

## 第四部分：`web/src/systems/CardSystem.ts`

```ts
import type Phaser from 'phaser';
import type {
  CardDef,
  CardModifier,
  ICardService,
  ModifierTarget,
  WeaponCategory,
  WeaponInstance,
  WeaponStats,
} from '../core/contracts';
import { allCards, getCardDef, getLevelScaling, getWeaponDef } from '../data';
import { getService } from '../core/services';
import { clamp } from '../core/util';

/** §7.3.2 封顶表 */
const CRIT_CHANCE_CAP = 0.75;
const LIFESTEAL_CAP = 0.3;
const ATTACK_SPEED_CAP = 5;
const EXECUTE_BELOW_CAP = 0.5;
const PROJECTILE_COUNT_MIN = 1;
const PROJECTILE_COUNT_MAX = 9;
const COMBO_HITS_MIN = 1;
const COMBO_HITS_MAX = 5;
/** §7.5：damage 硬上限 */
const DAMAGE_HARD_CAP = 200;

/** describe() 里需要把 0..1 显示成百分比的 target */
const PERCENT_TARGETS: readonly ModifierTarget[] = ['critChance', 'lifesteal', 'executeBelow'];
const SLOT_ORDER = ['primary', 'secondary'] as const;

/**
 * 卡牌数值结算、抽卡与文案。
 * 唯一写者是本系统；其他人只读 WeaponInstance.stats 快照。
 */
export class CardSystem implements ICardService {
  private readonly scene: Phaser.Scene;
  /** uid → 卡牌 params 合并结果；WeaponStats 里没有 params 字段，由这里按实例保存 */
  private readonly paramsByUid = new Map<string, Record<string, number>>();

  constructor(scene: Phaser.Scene) {
    this.scene = scene;
  }

  getDef(id: string): CardDef {
    return getCardDef(id);
  }

  /**
   * §7.3.2 数值结算管线：
   * base → 等级成长 → 锻造被动 → 卡牌 add → mul → set → 封顶 → flags/params。
   * 进化后的 defId 已由 UpgradeSystem 换好，这里只读当前 def。
   */
  computeStats(inst: WeaponInstance): WeaponStats {
    const def = getWeaponDef(inst.defId);
    const scaling = getLevelScaling();
    const levelIndex = clamp(inst.level - 1, 0, 2);
    const damageMul = scaling.damageMul[levelIndex] ?? 1;
    const attackSpeedMul = scaling.attackSpeedMul[levelIndex] ?? 1;

    const baseDamage = def.baseDamage * damageMul;
    const baseAttackSpeed = def.attackSpeed * attackSpeedMul;
    const forgeDamageMul = 1 + getService(this.scene, 'player').damagePctBonus();

    const values: Record<ModifierTarget, number> = {
      damage: baseDamage * forgeDamageMul,
      attackSpeed: baseAttackSpeed,
      range: def.range,
      comboHits: def.comboHits,
      projectileCount: def.projectile?.count ?? 1,
      pierce: def.projectile?.pierce ?? 0,
      spread: def.projectile?.spreadDeg ?? 0,
      critChance: def.critChance,
      critMult: def.critMult,
      executeBelow: 0,
      poisonPerHit: 0,
      poisonDuration: 0,
      lifesteal: 0,
      chain: 0,
      explodeRadius: 0,
      spinRadius: 0,
    };

    const flags: string[] = [];
    const params: Record<string, number> = {};
    const modifiers: CardModifier[] = [];

    for (const cardId of inst.cardIds) {
      const card = getCardDef(cardId);
      const applicable = card.modifiers.filter((mod) => this.modifierApplies(mod, def.category));
      // 没有任何一条 modifier 适配当前武器 → 整张卡跳过（含 flags/params）
      if (applicable.length === 0) continue;
      modifiers.push(...applicable);
      for (const flag of card.flags ?? []) {
        if (!flags.includes(flag)) flags.push(flag);
      }
      if (card.params) Object.assign(params, card.params);
    }

    for (const mod of modifiers) {
      if (mod.op === 'add') values[mod.target] += mod.value;
    }
    for (const mod of modifiers) {
      if (mod.op === 'mul') values[mod.target] *= mod.value;
    }
    for (const mod of modifiers) {
      if (mod.op === 'set') values[mod.target] = mod.value;
    }

    const snapshot: WeaponStats = {
      damage: Math.min(values.damage, DAMAGE_HARD_CAP),
      attackSpeed: Math.min(values.attackSpeed, ATTACK_SPEED_CAP),
      rangeTiles: values.range,
      comboHits: clamp(Math.round(values.comboHits), COMBO_HITS_MIN, COMBO_HITS_MAX),
      projectileCount: clamp(Math.round(values.projectileCount), PROJECTILE_COUNT_MIN, PROJECTILE_COUNT_MAX),
      pierce: Math.max(0, Math.round(values.pierce)),
      spreadDeg: values.spread,
      critChance: Math.min(values.critChance, CRIT_CHANCE_CAP),
      critMult: values.critMult,
      executeBelow: clamp(values.executeBelow, 0, EXECUTE_BELOW_CAP),
      poisonPerHit: Math.max(0, values.poisonPerHit),
      poisonDurationMs: Math.max(0, values.poisonDuration),
      lifesteal: Math.min(values.lifesteal, LIFESTEAL_CAP),
      chain: Math.max(0, Math.round(values.chain)),
      explodeRadiusPx: Math.max(0, values.explodeRadius),
      spinRadiusPx: Math.max(0, values.spinRadius),
      flags: Object.freeze([...flags]),
    };

    this.paramsByUid.set(inst.uid, Object.freeze({ ...params }));
    return Object.freeze(snapshot);
  }

  /** 供 WeaponSystem.buildSpec 读取合并后的 params（同名取最后一张卡） */
  getParams(inst: WeaponInstance): Record<string, number> {
    return this.paramsByUid.get(inst.uid) ?? {};
  }

  canInstall(inst: WeaponInstance, cardId: string): { ok: boolean; reason?: string } {
    const card = getCardDef(cardId);
    const def = getWeaponDef(inst.defId);

    if (!this.cardAppliesTo(card, def.category)) {
      return { ok: false, reason: '该武器不适用' };
    }
    if (inst.cardIds.length >= def.cardSlots) {
      return { ok: false, reason: '卡槽已满' };
    }
    if (!card.stackable && inst.cardIds.includes(cardId)) {
      return { ok: false, reason: '已装备同名卡牌' };
    }
    return { ok: true };
  }

  /**
   * 抽卡。epicBias 是 P3 内部扩展参数（精英房用），接口调用方只用前 3 个参数。
   * 候选规则：已解锁 → 适配「还有空卡槽的任一武器」类别 → 未装或可叠加。
   * 用两把武器的类别并集而不是只看当前武器，避免主手满槽时三选一/商店发不出去的卡。
   */
  rollOffers(
    count: number,
    installedCardIds: readonly string[],
    unlockedIds: readonly string[],
    epicBias = false,
  ): CardDef[] {
    const player = getService(this.scene, 'player');
    const categories = new Set<WeaponCategory>();
    for (const slot of SLOT_ORDER) {
      const inst = player.getWeapon(slot);
      if (!inst) continue;
      const def = getWeaponDef(inst.defId);
      if (inst.cardIds.length < def.cardSlots) categories.add(def.category);
    }
    if (categories.size === 0) return [];
    const categoryList = [...categories];

    const pool = allCards().filter((card) => {
      if (card.unlock !== undefined && !unlockedIds.includes(card.unlock)) return false;
      const fitsSomeWeapon = categoryList.some((category) => this.cardAppliesTo(card, category));
      if (!fitsSomeWeapon) return false;
      if (!card.stackable && installedCardIds.includes(card.id)) return false;
      return true;
    });

    const picked: CardDef[] = [];
    while (picked.length < count && pool.length > 0) {
      const total = pool.reduce((sum, card) => sum + this.weightOf(card, epicBias), 0);
      if (total <= 0) {
        const [fallback] = pool.splice(0, 1);
        if (fallback) picked.push(fallback);
        continue;
      }
      let roll = Math.random() * total;
      let pickedIndex = pool.length - 1;
      for (let i = 0; i < pool.length; i += 1) {
        roll -= this.weightOf(pool[i], epicBias);
        if (roll < 0) {
          pickedIndex = i;
          break;
        }
      }
      const [card] = pool.splice(pickedIndex, 1);
      if (card) picked.push(card);
    }
    return picked;
  }

  describe(card: CardDef): string {
    if (!card.description.includes('{value}')) return card.description;
    // 文案是「{value}%」时优先取百分比类 target，否则取第一条非百分比 target；
    // 避免「暴击率 +{value}%」误用第一条 critMult 之类的非百分比数值。
    const isPercentText = card.description.includes('{value}%');
    const candidates = isPercentText
      ? card.modifiers.filter((mod) => PERCENT_TARGETS.includes(mod.target))
      : card.modifiers;
    const shown = candidates[0] ?? card.modifiers[0];
    if (!shown) return card.description;
    const displayed = PERCENT_TARGETS.includes(shown.target)
      ? Math.round(shown.value * 100)
      : shown.value;
    return card.description.replace('{value}', String(displayed));
  }

  destroy(): void {
    this.paramsByUid.clear();
  }

  private modifierApplies(mod: CardModifier, category: WeaponCategory): boolean {
    return mod.weaponCategories === undefined || mod.weaponCategories.includes(category);
  }

  private cardAppliesTo(card: CardDef, category: WeaponCategory): boolean {
    return card.modifiers.some((mod) => this.modifierApplies(mod, category));
  }

  private weightOf(card: CardDef, epicBias: boolean): number {
    const base = Math.max(0, card.weight);
    // 精英房「稀有度加权 epic」：§7.3.5 未给具体倍率，这里统一 ×2（占位，策划可改）
    return epicBias && card.rarity === 'epic' ? base * 2 : base;
  }
}
```

```ts
import Phaser from 'phaser';
import type {
  DamageSpec,
  ElementType,
  IWeaponService,
  PlayerProjectileCfg,
  Vec2,
  WeaponDef,
  WeaponInstance,
  WeaponSlot,
  WeaponStats,
} from '../core/contracts';
import { bus } from '../core/events';
import { getService, setService } from '../core/services';
import { TILE_PX } from '../core/constants';
import { getCardDef, getWeaponDef } from '../data';
import { CardSystem } from './CardSystem';
import { UpgradeSystem } from './UpgradeSystem';
import { DropSystem } from './DropSystem';

/** §7.3.3 DamageSpec 模板 */
const DEFAULT_KNOCKBACK = 160;
const THIRD_HIT_ARC_MUL = 1.2;
const THIRD_HIT_KNOCKBACK_MUL = 1.5;
const HIT_STOP_MS = 40;
/** 连击左右换向的视觉偏移比例（不是伤害数值） */
const SWING_OFFSET_RATIO = 0.2;
const MIN_ATTACK_SPEED = 0.001;
const FULL_CIRCLE_DEG = 360;
const ELEMENT_TAGS: readonly ElementType[] = ['fire', 'poison', 'lightning', 'ice'];
/** buildSpec 里 params 缺省的兜底值（正式值都在 cards.json） */
const FALLBACK_EXECUTE_MULT = 2;
const FALLBACK_CHAIN_RANGE_PX = 96;
const FALLBACK_CHAIN_DAMAGE_MUL = 0.6;
const FALLBACK_BURST_PERCENT_MAX_HP = 0.2;

/**
 * 三种武器行为 + 卡牌即时生效。
 * 每帧只读 inst.stats，不允许把数值缓存进动画。
 */
export class WeaponSystem implements IWeaponService {
  private readonly scene: Phaser.Scene;
  private readonly card: CardSystem;
  private readonly instances = new Set<WeaponInstance>();
  private nextUid = 0;

  constructor(scene: Phaser.Scene, card: CardSystem) {
    this.scene = scene;
    this.card = card;
  }

  create(defId: string): WeaponInstance {
    const def = getWeaponDef(defId);
    this.nextUid += 1;
    const inst: WeaponInstance = {
      uid: `${def.id}#${this.nextUid}`,
      defId: def.id,
      level: 1,
      exp: 0,
      cardIds: [],
      evolved: false,
      stats: this.baseStats(def),
      nextAttackAt: 0,
      comboIndex: 0,
      comboExpireAt: 0,
    };
    this.instances.add(inst);
    this.recompute(inst);
    return inst;
  }

  installCard(inst: WeaponInstance, cardId: string): boolean {
    const verdict = this.card.canInstall(inst, cardId);
    if (!verdict.ok) {
      const reason = verdict.reason ?? '无法装备';
      bus.emit('card:rejected', { cardId, reason });
      bus.emit('ui:toast', { text: reason });
      return false;
    }

    inst.cardIds.push(cardId);
    this.recompute(inst);
    this.notifyState(inst);
    bus.emit('card:installed', { slot: this.slotOf(inst), cardId });

    const cardDef = getCardDef(cardId);
    getService(this.scene, 'audio').play('sfx_card');
    bus.emit('ui:toast', { text: `已装备·${cardDef.name}` });
    return true;
  }

  tryAttack(inst: WeaponInstance, origin: Vec2, aimRad: number, nowMs: number): boolean {
    if (nowMs < inst.nextAttackAt) return false;

    const stats = inst.stats;
    inst.nextAttackAt = nowMs + 1000 / Math.max(stats.attackSpeed, MIN_ATTACK_SPEED);

    const def = getWeaponDef(inst.defId);
    const spec = this.buildSpec(stats, def, this.card.getParams(inst));

    if (def.category === 'melee') {
      this.performMelee(inst, def, stats, spec, origin, aimRad, nowMs);
    } else {
      this.performRanged(def, stats, spec, origin, aimRad);
    }
    return true;
  }

  update(_dtMs: number, nowMs: number): void {
    for (const inst of this.instances) {
      if (inst.comboExpireAt > 0 && nowMs > inst.comboExpireAt) {
        inst.comboIndex = 0;
        inst.comboExpireAt = 0;
      }
    }
  }

  cooldownRatio(inst: WeaponInstance, nowMs: number): number {
    const stats = inst.stats;
    if (stats.attackSpeed <= 0) return 0;
    const cycleMs = 1000 / stats.attackSpeed;
    const remaining = Math.max(0, inst.nextAttackAt - nowMs);
    return Math.min(1, remaining / cycleMs);
  }

  recompute(inst: WeaponInstance): void {
    inst.stats = this.card.computeStats(inst);
  }

  /** UpgradeSystem 复用：发武器状态事件 */
  notifyState(inst: WeaponInstance): void {
    const upgrade = getService(this.scene, 'upgrade');
    bus.emit('weapon:state', {
      slot: this.slotOf(inst),
      defId: inst.defId,
      level: inst.level,
      exp: inst.exp,
      next: upgrade.expToNext(inst),
      canEvolve: upgrade.canEvolve(inst),
      cardIds: [...inst.cardIds],
      stats: inst.stats,
    });
  }

  slotOf(inst: WeaponInstance): WeaponSlot {
    const player = getService(this.scene, 'player');
    if (player.getWeapon('primary')?.uid === inst.uid) return 'primary';
    if (player.getWeapon('secondary')?.uid === inst.uid) return 'secondary';
    return 'primary';
  }

  destroy(): void {
    this.instances.clear();
  }

  private performMelee(
    inst: WeaponInstance,
    def: WeaponDef,
    stats: WeaponStats,
    spec: DamageSpec,
    origin: Vec2,
    aimRad: number,
    nowMs: number,
  ): void {
    const combat = getService(this.scene, 'combat');
    const audio = getService(this.scene, 'audio');
    const comboHits = Math.max(1, Math.round(stats.comboHits));
    const isSpin = stats.flags.includes('spin');

    if (nowMs > inst.comboExpireAt) inst.comboIndex = 0;

    let arcDeg = def.arcDeg ?? 90;
    let rangePx = stats.rangeTiles * TILE_PX;
    let swingRad = aimRad;

    if (isSpin) {
      arcDeg = FULL_CIRCLE_DEG;
      rangePx = stats.spinRadiusPx > 0 ? stats.spinRadiusPx : rangePx;
    } else {
      const isThirdHit = comboHits > 1 && inst.comboIndex === comboHits - 1;
      if (isThirdHit) arcDeg *= THIRD_HIT_ARC_MUL;
      const halfArcRad = ((arcDeg * Math.PI) / 180) / 2;
      const direction = inst.comboIndex % 2 === 0 ? -1 : 1;
      swingRad = aimRad + direction * halfArcRad * SWING_OFFSET_RATIO;
      if (isThirdHit) spec.knockback = (spec.knockback ?? 0) * THIRD_HIT_KNOCKBACK_MUL;
    }

    const hitCount = combat.meleeArc(origin, swingRad, rangePx, arcDeg, spec);

    // 魔焰剑：挥砍附带穿透剑气（def.projectile 存在时）
    if (def.projectile) {
      this.fireProjectiles(def, stats, spec, origin, aimRad);
    }

    if (isSpin || comboHits <= 1) {
      inst.comboIndex = 0;
      inst.comboExpireAt = 0;
    } else {
      inst.comboIndex = (inst.comboIndex + 1) % Math.max(1, comboHits);
      // 连击窗口从「下一次可攻击时刻」起算，而不是从本次攻击时刻起算：
      // 铁剑 1.0 次/秒（1000ms）> 620ms，若从本次攻击起算，三段连击永远进不到第 2 段。
      inst.comboExpireAt = inst.nextAttackAt + def.comboWindowMs;
    }

    audio.play('sfx_swing');
    if (hitCount > 0) combat.hitStop(HIT_STOP_MS);
  }

  private performRanged(
    def: WeaponDef,
    stats: WeaponStats,
    spec: DamageSpec,
    origin: Vec2,
    aimRad: number,
  ): void {
    if (!def.projectile) return;
    this.fireProjectiles(def, stats, spec, origin, aimRad);
    getService(this.scene, 'audio').play('sfx_shot');
  }

  private fireProjectiles(
    def: WeaponDef,
    stats: WeaponStats,
    spec: DamageSpec,
    origin: Vec2,
    aimRad: number,
  ): void {
    const proj = def.projectile;
    if (!proj) return;

    const combat = getService(this.scene, 'combat');
    const count = Math.max(1, Math.round(stats.projectileCount));
    const spreadRad = (stats.spreadDeg * Math.PI) / 180;
    // 长刃（range ×）对剑气同步生效：PlayerProjectileCfg 没有距离字段，
    // 用「有效距离 / 基础距离」等比放大存活时长，速度不变 → 射程变长。
    const rangeScale = def.category === 'melee' && def.range > 0
      ? stats.rangeTiles / def.range
      : 1;
    const lifeMs = proj.lifeMs * rangeScale;

    for (let i = 0; i < count; i += 1) {
      const offsetRad = count > 1 ? (i - (count - 1) / 2) * spreadRad : 0;
      const cfg: PlayerProjectileCfg = {
        from: origin,
        angleRad: aimRad + offsetRad,
        speed: proj.speed,
        lifeMs,
        radiusPx: proj.radiusPx,
        pierce: stats.pierce,
        spec: { ...spec },
      };
      if (def.category === 'elemental') cfg.trail = true;
      combat.spawnPlayerProjectile(cfg);
    }
  }

  /** §7.3.3 模板：P3 只决定性，暴击/处决/连锁/吸血/爆炸由 P2 结算 */
  private buildSpec(stats: WeaponStats, def: WeaponDef, params: Record<string, number>): DamageSpec {
    const spec: DamageSpec = {
      amount: stats.damage,
      element: this.resolveElement(def, stats.flags),
      critChance: stats.critChance,
      critMult: stats.critMult,
      knockback: DEFAULT_KNOCKBACK,
      tags: [...def.tags, ...stats.flags],
    };

    if (stats.executeBelow > 0) {
      spec.executeBelow = stats.executeBelow;
      spec.executeMult = params.executeMult ?? FALLBACK_EXECUTE_MULT;
    }

    if (stats.poisonPerHit > 0) {
      spec.status = {
        element: 'poison',
        dps: stats.poisonPerHit,
        durationMs: stats.poisonDurationMs,
        sourceId: 'weapon',
      };
    }

    if (stats.lifesteal > 0) spec.lifesteal = stats.lifesteal;

    if (stats.chain > 0) {
      spec.chain = {
        count: stats.chain,
        rangePx: params.chainRangePx ?? FALLBACK_CHAIN_RANGE_PX,
        damageMul: params.chainDamageMul ?? FALLBACK_CHAIN_DAMAGE_MUL,
      };
    }

    if (stats.flags.includes('plague_burst')) {
      spec.deathBurst = {
        radiusPx: stats.explodeRadiusPx,
        percentMaxHp: params.percentMaxHp ?? FALLBACK_BURST_PERCENT_MAX_HP,
        element: 'poison',
        requiresStatus: 'poison',
      };
    }

    return spec;
  }

  private resolveElement(def: WeaponDef, flags: readonly string[]): ElementType {
    const tags = new Set<string>([...def.tags, ...flags]);
    for (const element of ELEMENT_TAGS) {
      if (tags.has(element)) return element;
    }
    return 'physical';
  }

  /** create 首次填充用，随后立即被 CardSystem.computeStats 覆盖 */
  private baseStats(def: WeaponDef): WeaponStats {
    return {
      damage: def.baseDamage,
      attackSpeed: def.attackSpeed,
      rangeTiles: def.range,
      comboHits: def.comboHits,
      projectileCount: def.projectile?.count ?? 1,
      pierce: def.projectile?.pierce ?? 0,
      spreadDeg: def.projectile?.spreadDeg ?? 0,
      critChance: def.critChance,
      critMult: def.critMult,
      executeBelow: 0,
      poisonPerHit: 0,
      poisonDurationMs: 0,
      lifesteal: 0,
      chain: 0,
      explodeRadiusPx: 0,
      spinRadiusPx: 0,
      flags: [],
    };
  }
}

/** P3 唯一引导函数：注册 weapon / card / upgrade / drop，幂等，SHUTDOWN 清理 */
export function installBuild(scene: Phaser.Scene): void {
  const flag = 'p3.buildInstalled';
  if (scene.registry.get(flag) === true) return;
  scene.registry.set(flag, true);

  const card = new CardSystem(scene);
  const weapon = new WeaponSystem(scene, card);
  const upgrade = new UpgradeSystem(scene, weapon);
  const drop = new DropSystem(scene, card, upgrade);

  setService(scene, 'card', card);
  setService(scene, 'weapon', weapon);
  setService(scene, 'upgrade', upgrade);
  setService(scene, 'drop', drop);

  scene.events.once(Phaser.Scenes.Events.SHUTDOWN, () => {
    card.destroy();
    weapon.destroy();
    scene.registry.set(flag, false);
  });
}
```

```ts
import type Phaser from 'phaser';
import type {
  IUpgradeService,
  WeaponDef,
  WeaponExpResult,
  WeaponInstance,
} from '../core/contracts';
import { bus } from '../core/events';
import { getService } from '../core/services';
import { getWeaponDef } from '../data';
import type { WeaponSystem } from './WeaponSystem';

const MAX_LEVEL = 3;

/** 经验、升级、进化。进化只换 defId，已装卡牌全部保留。 */
export class UpgradeSystem implements IUpgradeService {
  private readonly scene: Phaser.Scene;
  private readonly weapon: WeaponSystem;

  constructor(scene: Phaser.Scene, weapon: WeaponSystem) {
    this.scene = scene;
    this.weapon = weapon;
  }

  expToNext(inst: WeaponInstance): number {
    const def = getWeaponDef(inst.defId);
    if (inst.evolved) return 0;
    if (inst.level < MAX_LEVEL) return def.expToLevel[inst.level - 1] ?? 0;
    return def.expToEvolve;
  }

  addExp(inst: WeaponInstance, amount: number): WeaponExpResult {
    if (inst.evolved) {
      return { levelUp: false, evolved: false, level: inst.level, exp: 0, next: 0 };
    }

    const def = getWeaponDef(inst.defId);
    let level = inst.level;
    let exp = inst.exp + Math.max(0, amount);
    let levelUp = false;

    while (level < MAX_LEVEL) {
      const need = def.expToLevel[level - 1] ?? 0;
      if (need <= 0 || exp < need) break;
      exp -= need;
      level += 1;
      levelUp = true;
    }

    // 满 Lv3 后经验只累计到进化所需值，HUD 进度条不会超过 100%
    if (level >= MAX_LEVEL) exp = Math.min(exp, def.expToEvolve);

    inst.level = level;
    inst.exp = exp;

    if (levelUp) {
      this.weapon.recompute(inst);
      getService(this.scene, 'audio').play('sfx_levelup');
    }
    this.weapon.notifyState(inst);

    return {
      levelUp,
      evolved: false,
      level,
      exp,
      next: this.expToNext(inst),
    };
  }

  canEvolve(inst: WeaponInstance): boolean {
    const def = getWeaponDef(inst.defId);
    return (
      !inst.evolved &&
      inst.level === MAX_LEVEL &&
      def.evolvesTo !== undefined &&
      def.expToEvolve > 0 &&
      inst.exp >= def.expToEvolve
    );
  }

  evolutionTarget(inst: WeaponInstance): WeaponDef | null {
    if (!this.canEvolve(inst)) return null;
    const def = getWeaponDef(inst.defId);
    return def.evolvesTo ? getWeaponDef(def.evolvesTo) : null;
  }

  evolve(inst: WeaponInstance): boolean {
    const target = this.evolutionTarget(inst);
    if (!target) return false;

    const from = inst.defId;
    const slot = this.weapon.slotOf(inst);

    inst.defId = target.id;
    inst.evolved = true;
    inst.level = MAX_LEVEL;
    inst.exp = 0;
    // 换形态后不继承旧形态的连击段，避免进化完第一刀直接算第 3 段
    inst.comboIndex = 0;
    inst.comboExpireAt = 0;
    this.weapon.recompute(inst);

    bus.emit('weapon:evolved', { slot, from, to: target.id });
    this.weapon.notifyState(inst);
    getService(this.scene, 'audio').play('sfx_evolve');
    return true;
  }
}
```

## 第七部分：`web/src/systems/DropSystem.ts`

```ts
import type Phaser from 'phaser';
import type {
  EnemyDef,
  IDropService,
  RewardContext,
  RewardOffer,
  RoomDef,
  WeaponInstance,
} from '../core/contracts';
import { bus } from '../core/events';
import { getService } from '../core/services';
import { allWeapons, getEconomy, getWeaponDef } from '../data';
import type { CardSystem } from './CardSystem';
import type { UpgradeSystem } from './UpgradeSystem';

/** §7.3.5 宝箱回血 */
const CHEST_HEAL_AMOUNT = 25;
/** 清房材料奖励；§7.3.5 未给具体数值，暂定 1 */
const ROOM_MATERIAL_AMOUNT = 1;
/** 精英房「稀有度加权 epic」；§7.3.5 未给倍率，暂定 ×2 */
const ELITE_EPIC_WEIGHT_MUL = 2;
/** 非卡奖励图标 key（需 P1 的 TextureFactory 提供同名占位图） */
const ICON_WEAPON_EXP = 'ui_icon_weapon_exp';
const ICON_MATERIAL = 'ui_icon_material';
const ICON_HEAL = 'ui_icon_heal';
const SLOT_ORDER = ['primary', 'secondary'] as const;

/**
 * 掉落与奖励：纯计算 + 应用。
 * 所有副作用都通过 ctx 里的 P2/P4 服务完成，P3 不直接写存档。
 */
export class DropSystem implements IDropService {
  private readonly scene: Phaser.Scene;
  private readonly card: CardSystem;
  private readonly upgrade: UpgradeSystem;
  private offerSeq = 0;

  constructor(scene: Phaser.Scene, card: CardSystem, upgrade: UpgradeSystem) {
    this.scene = scene;
    this.card = card;
    this.upgrade = upgrade;
  }

  killLoot(def: EnemyDef): { gold: number; material: number } {
    return {
      gold: def.gold,
      material: Math.random() < def.materialChance ? 1 : 0,
    };
  }

  roomRewards(room: RoomDef, _totalRooms: number): RewardOffer[] {
    if (room.type === 'shop') return this.shopOffers();
    if (room.type === 'boss') return [];

    const offers: RewardOffer[] = [];

    if (room.type === 'combat') {
      offers.push(...this.cardOffers(2));
      offers.push(this.randomNonCardOffer());
    } else if (room.type === 'elite') {
      offers.push(...this.cardOffers(1, true));
      offers.push(this.weaponExpOffer());
      offers.push(this.materialOffer(ROOM_MATERIAL_AMOUNT));
    } else if (room.type === 'treasure') {
      offers.push(...this.cardOffers(1));
      offers.push(this.materialOffer(ROOM_MATERIAL_AMOUNT));
      offers.push(this.healOffer(CHEST_HEAL_AMOUNT));
    }

    // 超出 rewardCount 时优先裁掉卡牌，保留非卡项（§7.5：非卡选项永远占 1 个位）
    while (offers.length > room.rewardCount) {
      let lastCardIndex = -1;
      for (let i = 0; i < offers.length; i += 1) {
        if (offers[i].kind === 'card') lastCardIndex = i;
      }
      offers.splice(lastCardIndex >= 0 ? lastCardIndex : offers.length - 1, 1);
    }
    while (offers.length < room.rewardCount) offers.push(this.fillerOffer(offers));

    // §7.3.4 方案 A：可进化时三选一里必有一个「进化」选项
    const evolution = this.evolutionOffer();
    if (evolution && offers.length > 0) {
      const nonCardIndex = offers.findIndex((offer) => offer.kind !== 'card');
      const replaceAt = nonCardIndex >= 0 ? nonCardIndex : offers.length - 1;
      offers[replaceAt] = evolution;
    }

    return offers;
  }

  chestReward(): RewardOffer {
    const roll = Math.floor(Math.random() * 3);
    if (roll === 0) return this.materialOffer(getEconomy().materialChest);
    if (roll === 1) {
      const [card] = this.cardOffers(1);
      if (card) return card;
      return this.materialOffer(getEconomy().materialChest);
    }
    return this.healOffer(CHEST_HEAL_AMOUNT);
  }

  shopOffers(): RewardOffer[] {
    const economy = getEconomy();
    const offers: RewardOffer[] = [];

    const [card] = this.cardOffers(1);
    if (card) {
      offers.push(this.withCost(card, economy.shop.cardPrice));
    }
    offers.push(this.withCost(this.healOffer(economy.shop.healAmount), economy.shop.healPrice));
    offers.push(
      this.withCost(this.materialOffer(economy.shop.materialAmount), economy.shop.materialPrice),
    );
    offers.push(
      this.withCost(
        this.weaponExpOffer(economy.shop.weaponExpAmount),
        economy.shop.weaponExpPrice,
      ),
    );

    return offers;
  }

  applyReward(offer: RewardOffer, ctx: RewardContext): { ok: boolean; reason?: string } {
    let result: { ok: boolean; reason?: string };
    // installCard 自己会发 ui:toast / card:rejected；这里标记一下，避免同一结果提示两遍
    let announced = false;

    switch (offer.kind) {
      case 'card': {
        const cardId = offer.cardId;
        if (!cardId) {
          result = { ok: false, reason: '奖励数据缺少卡牌 id' };
          break;
        }
        const target = this.findInstallTarget(cardId, ctx);
        if (!target.inst) {
          result = { ok: false, reason: target.reason };
          break;
        }
        const paid = this.charge(offer, ctx);
        if (!paid.ok) {
          result = paid;
          break;
        }
        const installed = ctx.weapon.installCard(target.inst, cardId);
        announced = true;
        if (installed) {
          result = { ok: true };
        } else {
          const verdict = ctx.card.canInstall(target.inst, cardId);
          result = { ok: false, reason: verdict.reason ?? '无法装备' };
        }
        break;
      }

      case 'weapon': {
        const weaponId = offer.weaponId;
        if (!weaponId) {
          result = { ok: false, reason: '奖励数据缺少武器 id' };
          break;
        }
        const evolutionOwner = this.findEvolutionOwner(weaponId, ctx);
        if (evolutionOwner) {
          const paid = this.charge(offer, ctx);
          if (!paid.ok) {
            result = paid;
            break;
          }
          result = ctx.upgrade.evolve(evolutionOwner)
            ? { ok: true }
            : { ok: false, reason: '进化失败' };
          break;
        }
        if (!allWeapons().some((weapon) => weapon.id === weaponId)) {
          result = { ok: false, reason: '目标武器不存在' };
          break;
        }
        const paid = this.charge(offer, ctx);
        if (!paid.ok) {
          result = paid;
          break;
        }
        ctx.player.equip('primary', weaponId);
        result = { ok: true };
        break;
      }

      case 'weaponExp': {
        const paid = this.charge(offer, ctx);
        if (!paid.ok) {
          result = paid;
          break;
        }
        ctx.upgrade.addExp(ctx.player.getWeapon('primary'), offer.amount ?? 0);
        result = { ok: true };
        break;
      }

      case 'material': {
        const paid = this.charge(offer, ctx);
        if (!paid.ok) {
          result = paid;
          break;
        }
        ctx.run.addMaterials(offer.amount ?? 0);
        result = { ok: true };
        break;
      }

      case 'heal': {
        const paid = this.charge(offer, ctx);
        if (!paid.ok) {
          result = paid;
          break;
        }
        // 满血也允许，返回 ok，由 UI 自己提示
        ctx.player.heal(offer.amount ?? 0);
        result = { ok: true };
        break;
      }

      case 'gold': {
        const paid = this.charge(offer, ctx);
        if (!paid.ok) {
          result = paid;
          break;
        }
        ctx.run.addGold(offer.amount ?? 0);
        result = { ok: true };
        break;
      }

      default:
        result = { ok: false, reason: '未知奖励类型' };
    }

    bus.emit('reward:applied', { offer, ok: result.ok });

    if (!announced) {
      if (result.ok) {
        bus.emit('ui:toast', { text: `获得·${offer.title}` });
      } else {
        const reason = result.reason ?? '无法获得';
        if (offer.kind === 'card' && offer.cardId) {
          bus.emit('card:rejected', { cardId: offer.cardId, reason });
        }
        bus.emit('ui:toast', { text: reason });
      }
    }

    return result;
  }

  private findInstallTarget(
    cardId: string,
    ctx: RewardContext,
  ): { inst: WeaponInstance | null; reason: string } {
    const primary = ctx.player.getWeapon('primary');
    const primaryVerdict = ctx.card.canInstall(primary, cardId);
    if (primaryVerdict.ok) return { inst: primary, reason: '' };

    const secondary = ctx.player.getWeapon('secondary');
    const secondaryVerdict = ctx.card.canInstall(secondary, cardId);
    if (secondaryVerdict.ok) return { inst: secondary, reason: '' };

    return {
      inst: null,
      reason: primaryVerdict.reason ?? secondaryVerdict.reason ?? '无法装备',
    };
  }

  private findEvolutionOwner(weaponId: string, ctx: RewardContext): WeaponInstance | null {
    for (const slot of SLOT_ORDER) {
      const inst = ctx.player.getWeapon(slot);
      const def = getWeaponDef(inst.defId);
      if (def.evolvesTo === weaponId && ctx.upgrade.canEvolve(inst)) return inst;
    }
    return null;
  }

  private charge(offer: RewardOffer, ctx: RewardContext): { ok: boolean; reason?: string } {
    if (offer.cost === undefined || offer.cost <= 0) return { ok: true };
    return ctx.run.spendGold(offer.cost)
      ? { ok: true }
      : { ok: false, reason: '金币不足' };
  }

  private cardOffers(count: number, epicBias = false): RewardOffer[] {
    const player = getService(this.scene, 'player');
    const installed: string[] = [];
    for (const slot of SLOT_ORDER) {
      const inst = player.getWeapon(slot);
      if (inst) installed.push(...inst.cardIds);
    }
    const unlocked = getService(this.scene, 'save').get().forge.unlocked;
    const cards = this.card.rollOffers(count, installed, unlocked, epicBias);
    return cards.map((card): RewardOffer => ({
      id: this.nextOfferId(),
      kind: 'card',
      title: card.name,
      description: this.card.describe(card),
      textureKey: card.textureKey,
      rarity: card.rarity,
      cardId: card.id,
    }));
  }

  private evolutionOffer(): RewardOffer | null {
    const player = getService(this.scene, 'player');
    for (const slot of SLOT_ORDER) {
      const inst = player.getWeapon(slot);
      if (!inst || !this.upgrade.canEvolve(inst)) continue;
      const target = this.upgrade.evolutionTarget(inst);
      if (target) {
        return {
          id: this.nextOfferId(),
          kind: 'weapon',
          title: `进化·${target.name}`,
          description: target.description,
          textureKey: target.textureKey,
          weaponId: target.id,
        };
      }
    }
    return null;
  }

  private randomNonCardOffer(): RewardOffer {
    const roll = Math.floor(Math.random() * 3);
    if (roll === 0) return this.weaponExpOffer();
    if (roll === 1) return this.materialOffer(ROOM_MATERIAL_AMOUNT);
    return this.healOffer(CHEST_HEAL_AMOUNT);
  }

  /** 三选一补位：优先选还没出现过的非卡 kind，保证选项互不重复 */
  private fillerOffer(existing: readonly RewardOffer[]): RewardOffer {
    const used = new Set(existing.map((offer) => offer.kind));
    if (!used.has('weaponExp')) return this.weaponExpOffer();
    if (!used.has('material')) return this.materialOffer(ROOM_MATERIAL_AMOUNT);
    if (!used.has('heal')) return this.healOffer(CHEST_HEAL_AMOUNT);
    // rewardCount > 3 的极端情况：只允许重复武器经验
    return this.weaponExpOffer();
  }

  private weaponExpOffer(amount?: number): RewardOffer {
    const value = amount ?? getEconomy().shop.weaponExpAmount;
    return {
      id: this.nextOfferId(),
      kind: 'weaponExp',
      title: '武器经验',
      description: `主手武器获得 ${value} 点经验`,
      textureKey: ICON_WEAPON_EXP,
      amount: value,
    };
  }

  private materialOffer(amount: number): RewardOffer {
    return {
      id: this.nextOfferId(),
      kind: 'material',
      title: '锻造材料',
      description: `获得 ${amount} 个锻造材料`,
      textureKey: ICON_MATERIAL,
      amount,
    };
  }

  private healOffer(amount: number): RewardOffer {
    return {
      id: this.nextOfferId(),
      kind: 'heal',
      title: '回复生命',
      description: `回复 ${amount} 点生命`,
      textureKey: ICON_HEAL,
      amount,
    };
  }

  private withCost(offer: RewardOffer, cost: number): RewardOffer {
    offer.cost = cost;
    return offer;
  }

  private nextOfferId(): string {
    this.offerSeq += 1;
    return `reward_${this.offerSeq}`;
  }
}
```

### 1. 合并顺序

1. 把三个 JSON 放进 `web/src/data/`，先跑 `npm run data:validate`。
2. 把四个 `.ts` 放进 `web/src/systems/`。
3. 确认 P2 的 `installCombat(this)` 在 `installBuild(this)` 之前调用（§4.6 固定顺序）。
4. `GameScene.create()` 里调用 `installBuild(this)`；P4 的 `installRun(this)` 放最后。

### 2. 与 P2 / P4 / P1 的对接清单（避免联调时互相等）

| 对接方 | 约定 |
|---|---|
| P2 `player.equip` | 内部必须调 `weapon.create(defId)`；装备完成后 `bus.emit('weapon:changed', { slot, defId })`。P3 的 `create()` 不发 `weapon:state`，因为此时实例还没挂到槽位上，`slotOf()` 无法可靠判断主/副手 |
| P2 攻击输入 | 鼠标按住时每帧调 `weapon.tryAttack(inst, playerPos, aimRad, scene.time.now)`；冷却在 P3 内部，P2 不需要再限速 |
| P2 `giveWeaponExp` | 主手 `upgrade.addExp(primary, amount)`，副手 `upgrade.addExp(secondary, amount * 0.5)`（§9.5） |
| P2 敌人死亡 | `drop.killLoot(def)` 与 `enemy:died` 里的 `gold/material` 是同一份数据，P4 只加一次，别重复结算 |
| P2 只读约束 | `WeaponInstance.stats`、`stats.flags`、`getParams()` 都是冻结/只读快照，P2 不得原地修改 |
| P4 奖励界面 | offer 来自 `drop.roomRewards / chestReward / shopOffers`；拿到结果后由 P4 发 `reward:offers { offers, source }`，选中后再发 `reward:chosen` 并调 `drop.applyReward(offer, ctx)`。**金币扣除已经在 P3 的 `charge()` 里做了，P4 不要再 `spendGold` 一次** |
| P4 `RewardContext` | `ctx.forgeUnlocked` P3 目前未使用（抽卡直接读 `save` 服务），照 §4 传值即可；`ctx.card` / `ctx.weapon` / `ctx.upgrade` / `ctx.run` 必须传真实服务 |
| P4 HUD | 监听 `weapon:state` 更新经验条/卡槽；首次装备后如需立刻显示，直接读 `getService('weapon')` 的实例并自己拼 `next/canEvolve`（或等第一次 `weapon:state`） |
| P4 商店 | `shopOffers()` 按 §7.3.5 返回 4 项（卡/回血/材料/武器经验）；§9.5 房间表写「三件商品」，UI 需要拍板显示 3 还是 4 |
| P1 贴图 | 需要补 `ui_icon_weapon_exp`、`ui_icon_material`、`ui_icon_heal` 三个非卡奖励占位图 |

### 3. 与文档口径不一致、需要团队确认的点

> ⚠ **最关键的一条**：铁剑 `attackSpeed 1.0`（1000ms 间隔）而 `comboWindowMs 620`。如果按字面「从本次攻击时刻起算」，下一次攻击时窗口早已过期，**三段连击永远只能打出第 1 段**。本文改为从「下一次可攻击时刻」起算，保留 620ms 作为「冷却好之后再犹豫多久就断连」的操作窗口。请策划确认这个解释，或把 `comboWindowMs` 调到 ≥ 攻击间隔。

| 点 | 本文处理 | 原因 |
|---|---|---|
| 连击窗口起算点 | 从 `nextAttackAt` 起算（冷却好后再等 620ms） | 否则 620ms < 1000ms，三段连击永远触发不了 |
| 抽卡类别过滤 | 取「还有空卡槽的武器」类别并集 | 只看当前主手会在主手满槽时继续发装不上的卡 |
| `hitStop` | 仅 `meleeArc` 命中数 > 0 时触发 | 空挥顿帧会让手感发黏 |
| 魔焰剑剑气数量 | `count: 1, spreadDeg: 0`，描述改为「一道穿透剑气」 | §9.1 弹道表是权威值；附录 D 文案「三向剑气」与表冲突 |
| 分裂弹 + 弹幕狂欢 | 手枪 `1 + 1 + 2 = 4` 发 | §9.2 数值相加的结果；§7.4 清单写的「5 发」与数据表不一致，**没有擅自改数据** |
| 进化后等级成长 | 进化形态仍套用 `levelScaling[2]`（魔焰剑 20 → 26 伤害） | §7.3.2 步骤 2 在步骤 3 之前且未排除进化态；若策划希望 20 为最终值，需要改 JSON 或给出新规则 |
| `applyReward` 的 `weapon` | 先判断是否是「进化奖励」，是则走 `upgrade.evolve`；否则 `equip('primary')` | 字面照 §7.3.5 直接 `equip` 会把 Lv3、已装卡牌全部清掉，与 §9.1「已装卡牌全部保留」冲突 |
| 精英房 epic 加权 | 内部权重 ×2 | §7.3.5 只说「稀有度加权 epic」，没给倍率；数值在 `ELITE_EPIC_WEIGHT_MUL` 一处 |
| 清房材料 / 回血数额 | 材料 1、回血 25、武器经验取 `economy.shop.weaponExpAmount`(60) | §7.3.5 只写了宝箱数值；这三项没有独立数据字段 |
| 手枪 0.08 后坐位移 | 未实现 | `IPlayerService` 没有位移接口；相机后坐属于 P2 的表现层 |
| 长刃对剑气的距离加成 | 已实现：近战弹道按 `rangeTiles / def.range` 等比放大 `lifeMs`，速度不变 | `PlayerProjectileCfg` 没有距离字段，只能改存活时长；扇形半径本来就读 `rangeTiles` |
| 非卡奖励图标 | `ui_icon_weapon_exp` / `ui_icon_material` / `ui_icon_heal` | 命名规范只给了卡牌/武器/敌人的 key；需要 P1 的 TextureFactory 补这三个占位图 |
